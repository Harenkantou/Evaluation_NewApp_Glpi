// excelImport.js
import * as XLSX from 'xlsx'
import { closeTicket, reopenTicket } from '@/services/ticketService'

async function importExcel(file) {
  const workbook = XLSX.read(await file.arrayBuffer())
  const rows = XLSX.utils.sheet_to_json(workbook.Sheets['Tickets'])
  
  for (const row of rows) {
    if (row.action === 'close') {
      await closeTicket({ 
        ticketId: row.ticket_id, 
        ticketName: row.name, 
        amount: row.montant 
      })
    }
  }
}
