<script setup>
import { ref } from 'vue'
import BoLayout from '@/components/backoffice/BoLayout.vue'
import { useGlpiStore } from '@/stores/glpi'
import { getTickets } from '@/services/glpiApi'
import { reopenTicket, closeTicket, cancelLastCost } from '@/services/ticketService'

const glpi = useGlpiStore()

// ─── État commun ─────────────────────────────────────────────
const loading = ref(false)
const message = ref('')
const logs = ref([])

// ─── 1️⃣ État SAISIE MANUELLE ────────────────────────────────
const manualTicket = ref('')
const manualMvt = ref('open')
const manualValue = ref('')

// ─── 2️⃣ État IMPORT FICHIER ─────────────────────────────────
const file = ref(null)
const setFile = (e) => { file.value = e.target.files?.[0] || null }

// ─── Parsing CSV ─────────────────────────────────────────────
function parseCSVLine(line) {
  const values = []
  let current = ''
  let inQuotes = false
  for (let i = 0; i < line.length; i++) {
    const ch = line[i]
    if (ch === '"') { inQuotes = !inQuotes; continue }
    if (ch === ',' && !inQuotes) { values.push(current.trim()); current = ''; continue }
    current += ch
  }
  values.push(current.trim())
  return values
}

function parseCSV(text) {
  const lines = text.replace(/\r/g, '').split('\n').filter(l => l.trim())
  if (!lines.length) return []
  const first = parseCSVLine(lines[0]).map(c => c.toLowerCase())
  const hasHeader = first.some(c => ['ticket', 'mvt', 'valeur'].includes(c))
  const data = hasHeader ? lines.slice(1) : lines
  return data.map(line => {
    const [ticket, mvt, value] = parseCSVLine(line)
    const num = value ? Number(value.replace(',', '.')) : null
    return {
      ticketRef: String(ticket || '').trim(),
      movement: String(mvt || '').trim().toLowerCase(),
      value: Number.isFinite(num) ? num : null
    }
  }).filter(r => r.ticketRef && r.movement)
}

// ─── Mapping Ref → ID GLPI ───────────────────────────────────
async function buildRefToIdMap() {
  const map = new Map()
  try {
    const token = await glpi.ensureToken()
    const tickets = await getTickets(token)
    if (!Array.isArray(tickets)) return map

    tickets.forEach(ticket => {
      if (!ticket || !ticket.id) return
      const content = String(ticket.content || '')
      const match = content.match(/Ref_Ticket:\s*(\d+)/i)
      if (match && match[1]) {
        map.set(String(match[1]), ticket.id)
      } else {
        map.set(String(ticket.id), ticket.id)
      }
    })
  } catch (err) {
    console.error('❌ Erreur buildRefToIdMap:', err)
  }
  return map
}

// ─── 🟢 FONCTION MÉTIER COMMUNE (comme traiter(a,b,c)) Si REACT ──────
// C'est ELLE qui est appelée depuis la saisie manuelle ET depuis le CSV !
async function traiterAction({ ticketRef, movement, value }, refToId) {
  // Récupérer l'ID GLPI
  let glpiId = refToId.get(ticketRef)
  if (!glpiId) {
    const directId = Number(ticketRef)
    if (Number.isInteger(directId) && directId > 0) glpiId = directId
  }
  if (!glpiId) {
    return { success: false, log: `⚠️ Ticket ref ${ticketRef} introuvable dans GLPI` }
  }

  const name = `Ticket #${glpiId} (ref ${ticketRef})`

  try {
    // 🟢 OPEN = Réouverture
    if (movement === 'open') {
      const result = await reopenTicket({
        ticketId: glpiId,
        ticketName: name,
        percent: value || 0
      })
      return {
        success: true,
        type: 'open',
        log: `🟢 ${name} OPEN ${value}% de ${result.baseCost.toFixed(2)}€ → +${result.cost.toFixed(2)}€`
      }
    }

    // 🔵 CLOSE = Clôture
    if (movement === 'close') {
      await closeTicket({
        ticketId: glpiId,
        ticketName: name,
        amount: value || 0
      })
      return {
        success: true,
        type: 'close',
        log: `🔵 ${name} CLOSE ${value}€`
      }
    }

    // ❌ CANCEL = Annulation
    if (movement === 'cancel') {
      const result = await cancelLastCost(glpiId)
      return {
        success: true,
        type: 'cancel',
        log: `❌ ${name} CANCEL → ${result.removed.costValue}€ supprimé`
      }
    }

    return { success: false, log: `⚠️ Mouvement inconnu : ${movement}` }
  } catch (err) {
    return { success: false, log: `❌ ${name} ${movement} échoué : ${err.message}` }
  }
}

// ─── 2️⃣ APPEL DEPUIS SAISIE MANUELLE ────────────────────────
async function handleClickManuel() {
  message.value = ''
  logs.value = []

  // Validation
  if (!manualTicket.value) {
    message.value = '❌ Veuillez saisir un ID de ticket'
    return
  }
  if (!manualMvt.value) {
    message.value = '❌ Veuillez choisir un mouvement'
    return
  }

  loading.value = true
  try {
    const refToId = await buildRefToIdMap()

    // ✅ Convertir la valeur (gère la virgule française)
    const rawValue = String(manualValue.value).replace(',', '.')
    const numValue = rawValue ? Number(rawValue) : null

    // 🎯 Appel de la fonction métier commune
    const result = await traiterAction({
      ticketRef: String(manualTicket.value).trim(),
      movement: manualMvt.value.toLowerCase(),
      value: numValue
    }, refToId)

    logs.value.push(result.log)
    message.value = result.success
      ? '✅ Action manuelle exécutée avec succès'
      : '❌ Action échouée'

    // Reset formulaire
    if (result.success) {
      manualTicket.value = ''
      manualValue.value = ''
    }
  } catch (e) {
    message.value = '❌ ' + (e.message || 'Erreur')
    console.error(e)
  } finally {
    loading.value = false
  }
}

// ─── 3️⃣ APPEL DEPUIS IMPORT CSV ─────────────────────────────
async function handleImport() {
  message.value = ''
  logs.value = []

  if (!file.value) {
    message.value = '❌ Veuillez choisir un fichier'
    return
  }

  loading.value = true
  try {
    const text = await file.value.text()
    const rows = parseCSV(text)
    if (!rows.length) throw new Error('CSV vide ou invalide')

    console.log('📂 Lignes parsées :', rows)

    const refToId = await buildRefToIdMap()
    const summary = { open: 0, close: 0, cancel: 0, ignored: 0 }

    // 🎯 Pour chaque ligne, appel de la MÊME fonction métier
    for (const row of rows) {
      const result = await traiterAction(row, refToId)
      logs.value.push(result.log)

      if (result.success) {
        summary[result.type]++
      } else {
        summary.ignored++
      }
    }

    message.value = `✅ Import : ${summary.open} open, ${summary.close} close, ${summary.cancel} cancel, ${summary.ignored} ignoré(s)`
  } catch (e) {
    message.value = '❌ ' + (e.message || 'Erreur')
    console.error(e)
  } finally {
    loading.value = false
  }
}
</script>
