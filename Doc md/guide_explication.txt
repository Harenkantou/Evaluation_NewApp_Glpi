<script setup>
import { ref } from 'vue'
import BoLayout from '@/components/backoffice/BoLayout.vue'
import { useGlpiStore } from '@/stores/glpi'
import { getTickets } from '@/services/glpiApi'
import { reopenTicket, closeTicket, cancelLastCost } from '@/services/ticketService'

const glpi = useGlpiStore()

// ─── État commun ─────────────────────────────────────────────
const loading = ref(false)
const message = ref('')
const logs = ref([])

// ─── 1️⃣ État SAISIE MANUELLE ────────────────────────────────
const manualTicket = ref('')
const manualMvt = ref('open')
const manualValue = ref('')

// ─── 2️⃣ État IMPORT FICHIER ─────────────────────────────────
const file = ref(null)
const setFile = (e) => { file.value = e.target.files?.[0] || null }

// ─── Parsing CSV (gère les guillemets) ───────────────────────
function parseCSVLine(line) {
  const values = []
  let current = ''
  let inQuotes = false
  for (let i = 0; i < line.length; i++) {
    const ch = line[i]
    if (ch === '"') { inQuotes = !inQuotes; continue }
    if (ch === ',' && !inQuotes) { values.push(current.trim()); current = ''; continue }
    current += ch
  }
  values.push(current.trim())
  return values
}

function parseCSV(text) {
  const lines = text.replace(/\r/g, '').split('\n').filter(l => l.trim())
  if (!lines.length) return []

  const first = parseCSVLine(lines[0]).map(c => c.toLowerCase())
  const hasHeader = first.some(c => ['ticket', 'mvt', 'valeur'].includes(c))
  const data = hasHeader ? lines.slice(1) : lines

  return data.map(line => {
    const [ticket, mvt, value] = parseCSVLine(line)
    const num = value ? Number(value.replace(',', '.')) : null
    return {
      ticketRef: String(ticket || '').trim(),
      movement: String(mvt || '').trim().toLowerCase(),
      value: Number.isFinite(num) ? num : null
    }
  }).filter(r => r.ticketRef && r.movement)
}

// ─── Mapping Ref → ID GLPI ───────────────────────────────────
async function buildRefToIdMap() {
  const map = new Map()
  
  try {
    const token = await glpi.ensureToken()
    const tickets = await getTickets(token)

    if (!Array.isArray(tickets)) {
      console.warn('⚠️ getTickets n\'a pas retourné un tableau:', tickets)
      return map
    }

    tickets.forEach(ticket => {
      if (!ticket || !ticket.id) return
      
      const content = String(ticket.content || '')
      const match = content.match(/Ref_Ticket:\s*(\d+)/i)
      
      if (match && match[1]) {
        map.set(String(match[1]), ticket.id)
      } else {
        // Fallback : utiliser l'ID GLPI directement
        map.set(String(ticket.id), ticket.id)
      }
    })

    console.log('🗺️ Map Ref → ID GLPI:', Object.fromEntries(map))
  } catch (err) {
    console.error('❌ Erreur buildRefToIdMap:', err)
  }
  
  return map
}

// ─── 🟢 FONCTION MÉTIER COMMUNE (équivalent traiter(a,b,c)) ──
// Cette fonction est appelée DEPUIS la saisie manuelle ET le CSV
async function traiterAction({ ticketRef, movement, value }, refToId, batchId = null) {
  // Récupérer l'ID GLPI
  let glpiId = refToId.get(ticketRef)
  
  if (!glpiId) {
    const directId = Number(ticketRef)
    if (Number.isInteger(directId) && directId > 0) glpiId = directId
  }
  
  if (!glpiId) {
    return { 
      success: false, 
      log: `⚠️ Ticket ref ${ticketRef} introuvable dans GLPI` 
    }
  }

  const name = `Ticket #${glpiId} (ref ${ticketRef})`

  try {
    // 🟢 OPEN = Réouverture (% du dernier coût)
    if (movement === 'open') {
      const result = await reopenTicket({
        ticketId: glpiId,
        ticketName: name,
        percent: value || 0,
        batchId
      })
      return {
        success: true,
        type: 'open',
        log: `🟢 ${name} OPEN ${value}% de ${result.baseCost.toFixed(2)}€ → +${result.cost.toFixed(2)}€`
      }
    }

    // 🔵 CLOSE = Clôture (montant fixe)
    if (movement === 'close') {
      await closeTicket({
        ticketId: glpiId,
        ticketName: name,
        amount: value || 0,
        batchId
      })
      return {
        success: true,
        type: 'close',
        log: `🔵 ${name} CLOSE ${value}€`
      }
    }

    // ❌ CANCEL = Annulation du dernier coût (groupé par batchId)
    if (movement === 'cancel') {
      const result = await cancelLastCost(glpiId)
      return {
        success: true,
        type: 'cancel',
        log: `❌ ${name} CANCEL → ${result.totalAmount}€ supprimé (${result.deletedCount} ligne(s))`
      }
    }

    return { 
      success: false, 
      log: `⚠️ Mouvement inconnu : ${movement}` 
    }
  } catch (err) {
    return { 
      success: false, 
      log: `❌ ${name} ${movement} échoué : ${err.message}` 
    }
  }
}

// ─── 2️⃣ APPEL DEPUIS LA SAISIE MANUELLE ─────────────────────
async function handleClickManuel() {
  message.value = ''
  logs.value = []

  // Validation
  if (!manualTicket.value) {
    message.value = '❌ Veuillez saisir un ID de ticket'
    return
  }

  if (!manualMvt.value) {
    message.value = '❌ Veuillez choisir un mouvement'
    return
  }

  // Validation valeur pour open/close
  if (manualMvt.value !== 'cancel' && (!manualValue.value || manualValue.value === '')) {
    message.value = `❌ La valeur est obligatoire pour ${manualMvt.value}`
    return
  }

  loading.value = true
  try {
    const refToId = await buildRefToIdMap()

    // ✅ Convertir la valeur (gère la virgule française)
    const rawValue = String(manualValue.value).replace(',', '.')
    const numValue = rawValue ? Number(rawValue) : null

    // 🎯 Appel de la fonction métier commune (PAS de batchId = source manual)
    const result = await traiterAction({
      ticketRef: String(manualTicket.value).trim(),
      movement: manualMvt.value.toLowerCase(),
      value: numValue
    }, refToId)

    logs.value.push(result.log)
    message.value = result.success
      ? '✅ Action manuelle exécutée avec succès'
      : '❌ Action échouée'

    // Reset formulaire si succès
    if (result.success) {
      manualTicket.value = ''
      manualValue.value = ''
    }
  } catch (e) {
    message.value = '❌ ' + (e.message || 'Erreur')
    console.error('❌ Erreur handleClickManuel:', e)
  } finally {
    loading.value = false
  }
}

// ─── 3️⃣ APPEL DEPUIS L'IMPORT CSV ───────────────────────────
async function handleImport() {
  message.value = ''
  logs.value = []

  if (!file.value) {
    message.value = '❌ Veuillez choisir un fichier'
    return
  }

  loading.value = true
  try {
    const text = await file.value.text()
    const rows = parseCSV(text)
    
    if (!rows.length) throw new Error('CSV vide ou invalide')

    console.log('📂 Lignes parsées :', rows)

    const refToId = await buildRefToIdMap()
    
    // ✅ Générer un batchId unique pour cet import CSV
    const importBatchId = `import-${Date.now()}`
    
    const summary = { open: 0, close: 0, cancel: 0, ignored: 0 }

    // 🎯 Pour chaque ligne, appel de la MÊME fonction métier
    for (const row of rows) {
      const result = await traiterAction(row, refToId, importBatchId)
      logs.value.push(result.log)

      if (result.success) {
        summary[result.type]++
      } else {
        summary.ignored++
      }
    }

    message.value = `✅ Import : ${summary.open} open, ${summary.close} close, ${summary.cancel} cancel, ${summary.ignored} ignoré(s)`
  } catch (e) {
    message.value = '❌ ' + (e.message || 'Erreur')
    console.error('❌ Erreur handleImport:', e)
  } finally {
    loading.value = false
  }
}

// ─── Réinitialiser ───────────────────────────────────────────
const resetLogs = () => {
  message.value = ''
  logs.value = []
}
</script>

<template>
  <BoLayout>

      <!-- ═══════════════════════════════════════════ -->

      <!-- ═══════════════════════════════════════════ -->
      <section class="card">

      <!-- ═══════════════════════════════════════════ -->
      <!-- SÉPARATEUR                                   -->

      <div class="separator">

      <!-- 2️⃣ IMPORT CSV                               -->
      <!-- ═══════════════════════════════════════════ -->
      <section class="card">

          <p class="hint-small">
            ⚠️ Les valeurs avec virgule décimale (ex: 16,5) doivent être entre guillemets
          </p>
        </details>
      </section>

      <!-- ═══════════════════════════════════════════ -->
      <!-- RÉSULTATS                                    -->
      <!-- ═══════════════════════════════════════════ -->
      <section v-if="message || logs.length" class="card results">
        <div class="results-header">
          <h2>📊 Résultats</h2>
          <button @click="resetLogs" class="btn-ghost">🗑️ Effacer</button>
        </div>

        <p 
          v-if="message"

        >
          {{ message }}
        </p>

        <ul v-if="logs.length" class="logs">
          <li v-for="(log, i) in logs" :key="i" :class="{
            'log-success': log.startsWith('🟢') || log.startsWith('🔵'),
            'log-cancel': log.startsWith('❌'),
            'log-warn': log.startsWith('⚠️')
          }">
            {{ log }}
          </li>
        </ul>
      </section>
    </div>
  </BoLayout>
</template>
