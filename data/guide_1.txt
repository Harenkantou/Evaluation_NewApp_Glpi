🟢 Fonction 1 : reopenTicket()
export async function reopenTicket({ ticketId, ticketName, percent, batchId }) {
  // 1. Récupère les closings du ticket
  // 2. Calcule le coût : lastBase × percent / 100
  // 3. Éclate le coût sur les items GLPI
  // 4. Sauvegarde avec un batchId commun
}

🎯 Utilisation N°1 : Dans le Kanban (drag & drop)

// Importation

// Appel dans la modale de réouverture
const confirmReopenWithPercent = async () => {
  const percent = Number(reopenPercentage.value) || 0
  
  // ✅ Appel de la fonction métier
  const result = await reopenTicket({
    ticketId: pendingReopen.value.id,
    ticketName: pendingReopen.value.name,
    percent
    // Pas de batchId → source = 'manual'
  })

  // Trace dans GLPI
  await addFollowup(
    token,
    pendingReopen.value.id,
    `Réouverture : ${percent}% de ${result.baseCost.toFixed(2)}€ → +${result.cost.toFixed(2)}€`
  )
}

  // 4. Sauvegarde 1 coût par item avec un batchId commun
}

🎯 Utilisation N°1 : Dans le Kanban (drag & drop)

const confirmClose = async () => {

    await addSolution(token, pendingClose.value.id, solutionText.value.trim())
  }

    amount
  })
}
